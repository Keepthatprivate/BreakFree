```javascript
// Capture Facebook Click ID from URL and store as _fbc cookie
(function() {
  const urlParams = new URLSearchParams(window.location.search);
  const fbclid = urlParams.get('fbclid');
  
  if (fbclid) {
    const timestamp = Date.now();
    const fbcValue = 'fb.1.' + timestamp + '.' + fbclid;
    const expiryDate = new Date();
    expiryDate.setDate(expiryDate.getDate() + 90);
    document.cookie = '_fbc=' + fbcValue + '; expires=' + expiryDate.toUTCString() + '; path=/; SameSite=Lax';
    localStorage.setItem('_fbc', fbcValue);
  }
})();

// Function to get Facebook cookies - call this before checkout
function getFacebookData() {
  const cookies = document.cookie.split(';').reduce((acc, cookie) => {
    const [key, value] = cookie.trim().split('=');
    if (key && value) acc[key] = value;
    return acc;
  }, {});
  
  return {
    fbc: cookies['_fbc'] || localStorage.getItem('_fbc') || null,
    fbp: cookies['_fbp'] || null,
    userAgent: navigator.userAgent,
    sourceUrl: window.location.href
  };
}
```

```javascript
const crypto = require('crypto');

async function sendFacebookPurchaseEvent(eventData) {
  const {
    email,
    phone,
    value,
    currency,
    eventId,
    clientIpAddress,
    clientUserAgent,
    eventSourceUrl,
    fbc,
    fbp
  } = eventData;

  const PIXEL_ID = process.env.FACEBOOK_PIXEL_ID;
  const ACCESS_TOKEN = process.env.FACEBOOK_ACCESS_TOKEN;

  if (!PIXEL_ID || !ACCESS_TOKEN) {
    console.error('Missing Facebook credentials');
    return { error: 'Missing credentials' };
  }

  // Hash email (required)
  const hashedEmail = email 
    ? crypto.createHash('sha256').update(email.toLowerCase().trim()).digest('hex')
    : null;

  // Hash phone if provided
  const hashedPhone = phone
    ? crypto.createHash('sha256').update(phone.replace(/\D/g, '')).digest('hex')
    : null;

  // Build user_data object with ALL parameters
  const userData = {};
  
  // Customer identifiers (at least one required)
  if (hashedEmail) userData.em = [hashedEmail];
  if (hashedPhone) userData.ph = [hashedPhone];
  
  // REQUIRED for website events
  if (clientIpAddress) userData.client_ip_address = clientIpAddress;
  if (clientUserAgent) userData.client_user_agent = clientUserAgent;
  
  // Facebook click/browser IDs (critical for attribution!)
  if (fbc) userData.fbc = fbc;
  if (fbp) userData.fbp = fbp;

  // Build the complete payload
  const payload = {
    data: [
      {
        event_name: 'Purchase',
        event_time: Math.floor(Date.now() / 1000),
        event_id: eventId, // REQUIRED for deduplication
        event_source_url: eventSourceUrl || 'https://YOUR_DOMAIN.com/checkout/success', // REQUIRED
        action_source: 'website',
        user_data: userData,
        custom_data: {
          currency: currency || 'USD',
          value: parseFloat(value) || 0
        }
      }
    ],
    access_token: ACCESS_TOKEN
  };

  // Log for debugging
  console.log('Sending to Facebook CAPI:', JSON.stringify(payload, null, 2));

  try {
    const response = await fetch(
      `https://graph.facebook.com/v18.0/${PIXEL_ID}/events`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      }
    );

    const result = await response.json();
    console.log('Facebook CAPI Response:', JSON.stringify(result, null, 2));
    
    if (result.error) {
      console.error('Facebook CAPI Error:', result.error);
    }
    
    return result;
  } catch (error) {
    console.error('Facebook CAPI Request Failed:', error);
    return { error: error.message };
  }
}
```
